<objective>
Implement the browser-side scenario engine in a new `crates/bootroom/web/scenario.js` ES module. The engine: locks the funnel, sequences the action list, accumulates per-action serial buffers (plus a secondary flat buffer for `after = "any"`), evaluates assertions (ANSI-stripped, line-buffered, substring or regex), enforces per-action + per-scenario timeouts, builds the JSONL-compatible transcript, and emits a single `ScenarioResult` WS frame to the server.

Phase 4's load-bearing browser surface. Most pitfalls cluster here (regex flavor drift, WS flush race, master.onWrite leak, `after = "any"` ordering). Each is addressed with a specific code pattern referenced inline.

Purpose: Provide the runtime side of the `ScenarioResult` handoff. 04-09 wires this module into `app.js` on detection of `?scenario=<name>`. 04-10 / 04-11 verify the full path end-to-end against real chromium.

    ```javascript
    /**
     * Browser-side scenario engine. Sole consumer is `bootroom run`'s URL-query
     * trigger (?scenario=<name>) wired in app.js (04-09).
     *
     * Architecture:
     *   1. Subscribe to master.onWrite. Each chunk is appended to:
     *      (a) per-action buffer Map<actionLabel, Uint8Array[]> — for
     *          assertions with `after = "<label>"`;
     *      (b) flat append-only Uint8Array[] — for `after = "any"`
     *          (Pitfall #5 line-arrival ordering).
     *   2. For each action in scenario.actions:
     *        - reset that action's per-action buffer (RUN-07 default);
     *        - currentLabel = label;
     *        - funnel.enqueue(b64ToBytes(action.bytes_b64), { pacingMs: 15 });
     *        - transcript.push({ts, type:'action_send', action, bytes_b64});
     *        - await Promise.race([
     *            evaluateAllAssertionsForAction(action),
     *            timeoutPromise(per-action timeout)
     *          ]);
     *        - emit assertion_result events to transcript;
     *        - record per-action verdict.
     *   3. After loop: build ScenarioResult JSON; ws.send(); poll
     *      bufferedAmount === 0 (Pitfall #2).
     *   4. Always run `finally`: funnel.unlockInput(); master.onWrite
     *      disposable .dispose() (Pitfall #4).
     *
     * Pitfalls handled inline (see 04-RESEARCH.md "Common Pitfalls"):
     *   #1 — regex flavor drift (Rust ∩ JS). Construct `new RegExp(pattern)`
     *        in a try/catch; on failure emit verdict='error' (defense-in-depth;
     *        Rust regex compile-check at config load makes this unreachable in
     *        practice).
     *   #2 — WS flush race. After ws.send(scenarioResultFrame), poll
     *        ws.bufferedAmount === 0 before resolving.
     *   #3 — funnel.enqueue is lock-agnostic; engine calls enqueue directly
     *        and never consults funnel.locked.
     *   #4 — master.onWrite Disposable is captured at subscription time and
     *        .dispose()'d in a `finally` block.
     *   #5 — `after = "any"` evaluates against the SECONDARY flat buffer
     *        (cross-action line order preserved), NOT the per-action Map.
     */

    // No imports — funnel / master / ws are injected via the deps argument.
    // bytesToB64 / b64ToBytes are duplicated inline (small; matches the
    // "no build step" constraint).

    function b64ToBytes(b64) {
      const bin = atob(b64);
      const out = new Uint8Array(bin.length);
      for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
      return out;
    }

    function bytesToB64(bytes) {
      let s = '';
      for (let i = 0; i < bytes.length; i++) s += String.fromCharCode(bytes[i]);
      return btoa(s);
    }

    function nowIsoUtc() {
      // Open Q3: UTC ISO 8601 with Z suffix. JS native toISOString() emits
      // exactly that format.
      return new Date().toISOString();
    }

    // ANSI strip — Pattern 3 / RUN-05.
    function stripAnsi(s) {
      return s.replace(/\x1b\[[0-9;]*[A-Za-z]/g, '');
    }

    function concatChunks(chunks) {
      let total = 0;
      for (const c of chunks) total += c.length;
      const out = new Uint8Array(total);
      let off = 0;
      for (const c of chunks) { out.set(c, off); off += c.length; }
      return out;
    }

    function decodeChunks(chunks) {
      return new TextDecoder('utf-8', { fatal: false }).decode(concatChunks(chunks));
    }

    function evaluate(rawChunks, assertion, atTimeout) {
      const raw = decodeChunks(rawChunks);
      const stripped = stripAnsi(raw);
      const lastNl = stripped.lastIndexOf('\n');
      const matchTarget = (atTimeout || lastNl === -1)
        ? stripped
        : stripped.slice(0, lastNl + 1);
      if (assertion.kind === 'contains') {
        return matchTarget.includes(assertion.pattern);
      }
      // 'regex': assertion._compiled was set at scenario start.
      try {
        return assertion._compiled.test(matchTarget);
      } catch (_e) {
        return false;
      }
    }

    function timeoutPromise(ms) {
      return new Promise((resolve) => setTimeout(() => resolve('timeout'), ms));
    }

    async function waitForFlush(ws) {
      // Pitfall #2. Poll bufferedAmount until 0 or until 5 s elapse (escape
      // hatch — bufferedAmount can lag on Chromium under heavy load; the
      // outer Rust timeout is the backstop).
      const deadline = Date.now() + 5_000;
      while (ws.bufferedAmount > 0 && Date.now() < deadline) {
        await new Promise((r) => setTimeout(r, 10));
      }
    }

    /**
     * Compute the per-action timeout: MAX(assertion.timeout_ms across
     * this action's assertions). If no assertions reference this action,
     * fallback 5_000 ms per Open Q5.
     */
    function perActionTimeoutMs(actionLabel, assertionList) {
      let max = 0;
      for (const a of assertionList) {
        if (a.after === actionLabel || a.after === 'any') {
          max = Math.max(max, a.timeout_ms || 5000);
        }
      }
      return max > 0 ? max : 5000;
    }

    /**
     * Pre-compile every regex assertion ONCE at scenario start. Mutates
     * the assertion list in place (adds `_compiled`); collects any compile
     * failures.
     */
    function compileRegexes(assertions) {
      const compileErrors = [];
      for (const a of assertions) {
        if (a.kind === 'regex') {
          try {
            a._compiled = new RegExp(a.pattern);
          } catch (e) {
            // Should be unreachable: Rust regex compile-check (04-02) is
            // the stricter engine. Defense-in-depth.
            compileErrors.push({ after: a.after, pattern: a.pattern, error: String(e) });
          }
        }
      }
      return compileErrors;
    }

    function getAssertionList(scenario) {
      // /api/config projects either `assert` (matching the TOML key) or
      // `assertions` (the Rust struct field). Accept both.
      return scenario.assertions || scenario.assert || [];
    }

    function buildErrorResult(scenario, error) {
      return {
        type: 'ScenarioResult',
        verdict: 'error',
        scenario: scenario && scenario.name ? scenario.name : '',
        started_at: nowIsoUtc(),
        ended_at: nowIsoUtc(),
        actions: [],
        transcript: [],
        error,
      };
    }

    export async function runScenario(scenario, actions, deps) {
      const { ws, funnel, master } = deps;

      // Open Q2 defense-in-depth: malformed scenario object.
      if (!scenario || typeof scenario.name !== 'string' || !Array.isArray(scenario.actions)) {
        const frame = buildErrorResult(scenario, 'scenario object malformed');
        try { ws.send(JSON.stringify(frame)); await waitForFlush(ws); } catch (_e) {}
        return;
      }

      const assertions = getAssertionList(scenario);
      const compileErrors = compileRegexes(assertions);
      if (compileErrors.length > 0) {
        const frame = buildErrorResult(
          scenario,
          'regex compile failed: ' + JSON.stringify(compileErrors)
        );
        try { ws.send(JSON.stringify(frame)); await waitForFlush(ws); } catch (_e) {}
        return;
      }

      const actionsByLabel = new Map(actions.map(a => [a.label, a]));
      const startedAt = nowIsoUtc();
      const transcript = [];               // TranscriptEvent[]
      const actionResults = [];            // [{label, verdict, assertions:[{kind,pattern,verdict}]}]

      // Per-action buffer (Pattern 2 + RUN-07).
      const buffers = new Map();           // label -> Uint8Array[]
      // Secondary flat buffer for `after = "any"` (Pitfall #5).
      const flat = [];                     // Uint8Array[]
      let currentLabel = null;

      // Subscribe to master.onWrite. Pitfall #4: capture disposable.
      const disposable = master.onWrite(([bytes, _ack]) => {
        if (!bytes || bytes.length === 0) return;
        if (currentLabel === null) return;
        // Per-action.
        const chunks = buffers.get(currentLabel) || [];
        chunks.push(new Uint8Array(bytes));
        buffers.set(currentLabel, chunks);
        // Secondary flat.
        flat.push(new Uint8Array(bytes));
        // Transcript chunk event.
        transcript.push({
          ts: nowIsoUtc(),
          type: 'serial_chunk',
          action: currentLabel,
          bytes_b64: bytesToB64(bytes),
        });
      });

      let scenarioVerdict = 'pass';   // demoted to 'fail' / 'timeout' / 'error' as failures accumulate
      let scenarioError = null;

      // Outer per-scenario timeout (RUN-06).
      const scenarioBudgetMs = scenario.timeout_ms || 30_000;
      const scenarioTimeoutKey = Symbol('scenarioTimeout');

      try {
        funnel.lockInput();

        const inner = async () => {
          for (const label of scenario.actions) {
            const action = actionsByLabel.get(label);
            if (!action) {
              actionResults.push({
                label, verdict: 'error',
                assertions: [],
                error: 'unknown action label',
              });
              scenarioVerdict = 'error';
              scenarioError = `unknown action '${label}'`;
              continue;
            }

            // Reset per-action buffer (RUN-07 default).
            buffers.set(label, []);
            currentLabel = label;

            // Enqueue bytes (Pitfall #3: lock-agnostic by design).
            const bytes = b64ToBytes(action.bytes_b64 || '');
            transcript.push({
              ts: nowIsoUtc(),
              type: 'action_send',
              action: label,
              bytes_b64: action.bytes_b64 || '',
            });
            if (bytes.length > 0) {
              funnel.enqueue(bytes, { pacingMs: 15 });
            }

            // Compute per-action timeout (Open Q5).
            const actionTimeoutMs = perActionTimeoutMs(label, assertions);

            // Collect assertions whose `after` matches this label or "any".
            const relevant = assertions.filter(a => a.after === label || a.after === 'any');

            // Poll until either all relevant assertions pass OR timeout.
            // Each assertion is "sticky" — once it passes, it stays passed
            // for that action.
            const passed = new Map(relevant.map(a => [a, false]));
            const pollDeadline = Date.now() + actionTimeoutMs;
            let pollResult = null;

            while (Date.now() < pollDeadline) {
              for (const a of relevant) {
                if (passed.get(a)) continue;
                // Choose buffer per Pitfall #5.
                const target = a.after === 'any' ? flat : (buffers.get(label) || []);
                if (evaluate(target, a, false)) passed.set(a, true);
              }
              if ([...passed.values()].every(Boolean)) {
                pollResult = 'all-passed';
                break;
              }
              await new Promise(r => setTimeout(r, 25));
            }
            if (pollResult === null) pollResult = 'timeout';

            // Final pass at timeout — include partial trailing line (RUN-05).
            if (pollResult === 'timeout') {
              for (const a of relevant) {
                if (passed.get(a)) continue;
                const target = a.after === 'any' ? flat : (buffers.get(label) || []);
                if (evaluate(target, a, true)) passed.set(a, true);
              }
            }

            // Build per-assertion verdict list for this action.
            const perAssert = relevant.map(a => {
              const v = passed.get(a) ? 'pass' : 'fail';
              transcript.push({
                ts: nowIsoUtc(),
                type: 'assertion_result',
                action: label,
                kind: a.kind,
                pattern: a.pattern,
                verdict: v,
              });
              return { kind: a.kind, pattern: a.pattern, verdict: v };
            });
            const allActionAssertionsPassed = perAssert.every(p => p.verdict === 'pass');
            const actionVerdict = allActionAssertionsPassed
              ? (pollResult === 'timeout' && relevant.length === 0 ? 'pass' : 'pass')
              : (pollResult === 'timeout' ? 'timeout' : 'fail');

            actionResults.push({
              label, verdict: actionVerdict,
              assertions: perAssert,
            });
            if (actionVerdict !== 'pass') {
              if (scenarioVerdict === 'pass') scenarioVerdict = actionVerdict;
            }
            // Reset currentLabel between actions so trailing-arrival chunks
            // do NOT pollute the next action's buffer.
            currentLabel = null;
          }
        };

        await Promise.race([
          inner(),
          timeoutPromise(scenarioBudgetMs).then(() => {
            scenarioVerdict = 'timeout';
            scenarioError = `scenario timeout after ${scenarioBudgetMs}ms`;
          }),
        ]);

      } catch (e) {
        scenarioVerdict = 'error';
        scenarioError = String(e);
      } finally {
        try { disposable && disposable.dispose && disposable.dispose(); } catch (_e) {}
        try { funnel.unlockInput(); } catch (_e) {}
      }

      const endedAt = nowIsoUtc();
      const transcriptResultEvent = {
        ts: endedAt,
        type: 'scenario_result',
        verdict: scenarioVerdict,
        actions: actionResults,
      };
      transcript.push(transcriptResultEvent);

      const frame = {
        type: 'ScenarioResult',
        verdict: scenarioVerdict,
        scenario: scenario.name,
        started_at: startedAt,
        ended_at: endedAt,
        actions: actionResults,
        transcript,
        error: scenarioError,
      };

      try {
        ws.send(JSON.stringify(frame));
        await waitForFlush(ws);   // Pitfall #2
      } catch (e) {
        // Best-effort. The Rust outer timeout in run_cmd is the backstop.
        // Log to console for headed-debug runs.
        console.warn('[scenario] ws.send(ScenarioResult) failed:', e);
      }
    }
    ```

## Trust Boundaries

The engine runs in the browser sandbox. The `scenario` and `actions` objects flow from `/api/config` which is server-projected from `bootroom.toml` (operator-controlled). The bytes injected via `funnel.enqueue` originate from the same trusted config.
