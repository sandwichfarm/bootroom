    ```rust
    //! as `bootroom serve`. Any change that introduces a separate run-mode
    //! router breaks this test.

    use bootroom::{build_router, AppState};
    use axum::body::Body;
    use axum::http::{Request, StatusCode};
    use std::{path::PathBuf, sync::Arc};
    use tempfile::NamedTempFile;
    use tower::ServiceExt;

    #[tokio::test]
    async fn run_router_reuses_serve_router() {
        let kernel = NamedTempFile::new().expect("tempfile");
        let state = Arc::new(AppState::new_for_test(kernel.path().to_path_buf(), None));
        let app = build_router(state);

        for path in ["/", "/api/config", "/api/kernel/info"] {
            let res = app.clone()
                .oneshot(Request::builder().uri(path).body(Body::empty()).unwrap())
                .await
                .expect("oneshot");
            // Each route exists — even if /api/kernel/info errors because
            // the kernel is just a tempfile (size > 0 / mtime probe works),
            // the response status MUST NOT be 404. The point of this test is
            // "no separate codepath," not "route returns 200."
            assert_ne!(res.status(), StatusCode::NOT_FOUND, "route {path} missing");
        }

        // /ws expects an upgrade header. A plain GET returns 426 Upgrade Required
        // or 400 Bad Request; both prove the route exists.
        let res = app
            .oneshot(Request::builder().uri("/ws").body(Body::empty()).unwrap())
            .await
            .expect("oneshot /ws");
        assert_ne!(res.status(), StatusCode::NOT_FOUND, "/ws route missing");
    }
    ```

    ```rust
    //! 04-10 — RUN-08 pin: --log-file writes JSONL transcript.
    //!
    //! Sub-second test (no chromium-roundtrip required): force a chromium
    //! discovery failure via $CHROMIUM=/nonexistent so the driver exits 3
    //! BEFORE attempting a real headless run, but AFTER writing the
    //! scenario_start preamble. Confirms (a) the file is created at the
    //! supplied path, (b) the first line is a valid scenario_start
    //! TranscriptEvent, (c) every line is valid JSON parsing to a known
    //! TranscriptEvent variant.

    use bootroom::transcript::TranscriptEvent;
    use std::{io::Write, process::Command};
    use tempfile::TempDir;

    #[test]
    fn run_writes_scenario_start_event_to_log_file() {
        let exe = env!("CARGO_BIN_EXE_bootroom");
        let tmp = TempDir::new().expect("tempdir");

        // 1. Create a valid kernel placeholder.
        let kernel = tmp.path().join("Image");
        std::fs::write(&kernel, b"not actually a kernel").expect("write kernel");

        // 2. Create a minimal bootroom.toml with one scenario.
        let cfg = tmp.path().join("bootroom.toml");
        std::fs::write(&cfg, b"\
schema_version = 1

[[action]]
label = \"noop\"
bytes = ''

[[scenario]]
name = \"smoke\"
actions = [\"noop\"]
timeout_ms = 1000
").expect("write toml");

        let log_path = tmp.path().join("log.jsonl");

        // 3. Force chromium discovery to fail so we hit exit 3 quickly.
        let out = Command::new(exe)
            .env("CHROMIUM", "/nonexistent")
            .args(["run",
                   "--kernel", kernel.to_str().unwrap(),
                   "--config", cfg.to_str().unwrap(),
                   "--scenario", "smoke",
                   "--log-file", log_path.to_str().unwrap()])
            .output()
            .expect("spawn bootroom run");

        // Exit code MAY be 3 (chromium missing) or 2 (other config error).
        // We don't gate on the exit code; we gate on the log content.
        let _ = out;

        // 4. The log file must exist with at least one JSON line.
        let content = std::fs::read_to_string(&log_path).unwrap_or_default();
        assert!(!content.is_empty(),
            "expected --log-file to be created with at least scenario_start;\nstderr was:\n{}",
            String::from_utf8_lossy(&out.stderr));

        // 5. Every line must be a valid TranscriptEvent.
        let mut found_scenario_start = false;
        for (i, line) in content.lines().enumerate() {
            let ev: TranscriptEvent = serde_json::from_str(line)
                .unwrap_or_else(|e| panic!("line {i} not valid TranscriptEvent JSON: {e}\nline: {line}"));
            if matches!(ev, TranscriptEvent::ScenarioStart { .. }) {
                found_scenario_start = true;
            }
        }
        assert!(found_scenario_start,
            "expected at least one scenario_start event; got:\n{content}");
    }
    ```
