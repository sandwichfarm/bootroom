    The file is a single ES module. It performs:

    1. **Kernel-info fetch + header population.** On script load:
       ```javascript
       async function loadKernelInfo() {
         try {
           const res = await fetch('/api/kernel/info');
           if (!res.ok) throw new Error('HTTP ' + res.status);
           const info = await res.json();
           document.getElementById('k-path').textContent  = info.path;
           document.getElementById('k-size').textContent  = humanBytes(info.size);
           document.getElementById('k-mtime').textContent = isoLocal(info.mtime);
           document.getElementById('k-sha').textContent   = info.sha256_prefix;
           // Update page title with kernel basename
           const base = info.path.split('/').pop() || info.path;
           document.title = 'bootroom — ' + base;
         } catch (e) {
           console.error('kernel-info fetch failed:', e);
           for (const id of ['k-path','k-size','k-mtime','k-sha']) {
             const el = document.getElementById(id);
             if (el) el.textContent = 'ERR';
           }
         }
       }
       ```

       Helpers (define inline):
       - `humanBytes(n)`: per UI-SPEC, IEC binary units; one decimal place when ≥10, two below. So `humanBytes(13049856)` → `"12.4 MiB"`.
       - `isoLocal(epochSec)`: per UI-SPEC, ISO-8601 local timezone with second precision. So `isoLocal(1715961128)` → `"2026-05-17T14:32:08-07:00"` (local TZ). Use `new Date(ms).toISOString()` then convert to local; or `Intl.DateTimeFormat` with timeZoneName. Simplest correct implementation: build the local ISO string manually with `getFullYear()`, `getMonth()+1`, etc., padded to 2 digits, plus the local offset via `getTimezoneOffset()`.

    2. **Status pill state machine.** Per Pattern 5:
       ```javascript
       const pill = document.getElementById('status');
       function setPill(state) {
         pill.dataset.state = state;
         pill.innerHTML = '<span aria-hidden="true">●</span> ' + state;
       }
       // initial state already set in HTML to LOADING; this is the JS-controlled transition machinery.
       ```

    3. **xterm + xterm-pty mount** (Pattern 4, verbatim from reference but with vendored libs):
       ```javascript
       const xterm = new Terminal();
       xterm.open(document.getElementById('terminal'));

       // Phase 1: input deliberately no-op; Phase 2 wires through /ws
       xterm.attachCustomKeyEventHandler(() => false);

       const { master, slave } = openpty();
       xterm.loadAddon(master);

       // window.Module was set by /assets/qemu/module.js (the QEMU argv)
       Module.pty = slave;
       Module.mainScriptUrlOrBlob = location.origin + '/assets/qemu/out.js';
       ```

    4. **Kernel byte injection + emscripten init.** Before calling `initEmscriptenModule`, fetch `/kernel` and write the bytes into `Module.FS` so the QEMU argv `-kernel /pack/Image` finds the right file. Spike A may later confirm an in-place swap path; for Phase 1 the initial-load flow is "fetch + writeFile":
       ```javascript
       async function bootGuest() {
         // Hook lifecycle BEFORE initEmscriptenModule so we don't miss early callbacks
         Module.onRuntimeInitialized = () => {
           if (self.crossOriginIsolated) setPill('RUNNING');
         };
         Module.onExit  = (code) => setPill('HALTED');
         Module.onAbort = (what) => setPill('HALTED');

         // We also need to inject the kernel bytes BEFORE QEMU starts. Hook
         // preRun to do that — emscripten calls preRun callbacks after Module.FS
         // is mounted but before main runs.
         Module.preRun = Module.preRun || [];
         Module.preRun.push(async () => {
           // emscripten preRun supports async functions if returned
           try {
             const res = await fetch('/kernel');
             if (!res.ok) throw new Error('kernel fetch HTTP ' + res.status);
             const bytes = new Uint8Array(await res.arrayBuffer());
             Module.FS.writeFile('/pack/Image', bytes);
           } catch (e) {
             console.error('kernel inject failed:', e);
             // Render a one-liner into the terminal so the user sees something
             slave.write('[bootroom] Failed to load kernel from /kernel: ' + e.message + '\r\n');
           }
         });

         // Dynamic import of the emscripten glue
         const mod = await import('/assets/qemu/out.js');
         const initEmscriptenModule = mod.default;
         const instance = await initEmscriptenModule(Module);

         // Reference's poll patch — copy verbatim to avoid PTY hangs.
         const oldPoll = Module.TTY.stream_ops.poll;
         const pty = Module.pty;
         Module.TTY.stream_ops.poll = function(stream, timeout) {
           if (!pty.readable) {
             return (pty.readable ? 1 : 0) | (pty.writable ? 4 : 0);
           }
           return oldPoll.call(stream, timeout);
         };
       }
       ```

       Notes on the preRun pattern:
       - emscripten's `preRun` runs synchronously by default; async support exists via `addRunDependency`/`removeRunDependency`. If the simple `async () => { ... }` push doesn't work in practice (it depends on the emscripten build flags qemu-wasm used), the fallback is to fetch the bytes BEFORE calling `initEmscriptenModule`, store on `Module._pendingKernel = bytes`, and push a synchronous `preRun` that calls `Module.FS.writeFile('/pack/Image', Module._pendingKernel)`.
       - If `/pack/` doesn't exist in the emscripten FS at preRun time (the data pack might mount it later), use `Module.FS.mkdirTree('/pack')` before `writeFile`.
       - Plan 01-08 (Spike B) and the in-browser smoke verification will exercise this path end-to-end.

    5. **Boot orchestration.** At the bottom of the module:
       ```javascript
       loadKernelInfo();
       bootGuest().catch(e => {
         console.error('boot failed:', e);
         setPill('HALTED');
         slave.write('[bootroom] Boot failed: ' + e.message + '\r\n');
       });
       ```

    Module scope notes:
    - `window.Terminal`, `window.openpty`, and `window.Module` are populated by the classic-script tags loaded by index.html BEFORE the module loads. The module reads them as globals.
    - Browsers defer `type="module"` scripts by default, so the classic scripts have run by the time the module starts. (Same property that makes the SAB probe safe.)
