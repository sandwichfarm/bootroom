    ```rust
    //! Common test helpers — spawn an ephemeral-port bootroom server.

    use bootroom::{build_router, AppState};
    use std::{path::PathBuf, sync::Arc};
    use tokio::net::TcpListener;

    pub struct TestServer {
        pub base_url: String,
        _handle: tokio::task::JoinHandle<()>,
    }

    /// Spawn a bootroom HTTP server on an ephemeral port.
    /// The server runs on the same tokio runtime as the test.
    pub async fn spawn(kernel: PathBuf, assets_dir: Option<PathBuf>) -> TestServer {
        let state = Arc::new(AppState::new(kernel, assets_dir));
        let app = build_router(state);
        let listener = TcpListener::bind(("127.0.0.1", 0)).await.expect("bind 0");
        let addr = listener.local_addr().expect("local_addr");
        let handle = tokio::spawn(async move {
            axum::serve(listener, app).await.expect("axum::serve");
        });
        TestServer {
            base_url: format!("http://{}", addr),
            _handle: handle,
        }
    }

    /// Write a tempfile with `bytes`; caller deletes via `tempfile::NamedTempFile` API.
    pub fn write_kernel_tempfile(bytes: &[u8]) -> tempfile::NamedTempFile {
        use std::io::Write;
        let mut f = tempfile::NamedTempFile::new().expect("tempfile");
        f.write_all(bytes).expect("write");
        f
    }
    ```

    Notes:
    - Returning `TestServer` keeps the JoinHandle alive for the duration of the test scope (drop = abort). For tests that need to assert on server-side panics, the test can `.await` the handle; for normal tests, we just want the server alive while reqwest probes.
    - The `_handle` field is prefixed with `_` so the dead-code lint allows it.
